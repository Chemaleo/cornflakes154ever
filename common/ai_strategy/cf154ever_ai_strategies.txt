PRC_dont_join_SOV = {
	enable = {
		original_tag = PRC
	}
	
	abort = {
		OR = {
			AND = {
				CHI = {
					OR = {
						exists = no
						is_subject_of = JAP
					}
				}
				SOV = { NOT = { has_war_with = GER } }
			}
			is_historical_focus_on = no
			date > 1946.1.1
		}
	}
	
	reversed = yes
	
	ai_strategy = {
		type = alliance
		id = SOV
		value = -200
	}
}
